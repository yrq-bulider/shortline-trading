## 1. 背景与目标

### 1.1 现状

`筛选明日股票_v1.py` v1.3 已完成:
- 4 维评分(技术 30 / 业绩 25 / 资金 20 / 消息 25)
- 消息面拆 4 子源:ann 30% + newsfeed 30% + hsgt 20% + lhb 20%
- 证伪门 9 条
- v1.3:涨停池 + 大盘情绪门
- 已有"机构"信号:主力资金 5 日、北向 3 日、龙虎榜机构席位、公告减持关键词

**用户反馈**(2026-06-15):
- 4 维评分"不是很准",目标胜率 70% 仍未达成(当前 ~50%)
- 希望加入"主力不小心流出/机构流出"的**深层次信号**
- 期望一次性大改 (v2.0),不分批

### 1.2 目标

把 4 类机构流出信号纳入 v2.0,胜率从 ~50% 推向 ≥70%:

1. **大宗交易折价 + 卖方席位** — 机构原始股东减持的"最后一步"
2. **主力分单细粒度** — 现有"主力净流入"拆特大/大/中/小单,识别"机构吸筹 vs 派发给散户"
3. **融资融券余额变化** — 杠杆资金撤退的日级信号
4. **股东户数变化 + 解禁压力** — 筹码集中度 + 未来卖压(准但有滞后)

### 1.3 非目标

- ❌ 不做实盘对接(只输出 TOP3 + 候选名单)
- ❌ 不改证伪门 9 条触发阈值(只新增 3 条与机构流出相关的)
- ❌ 不替换现有 baostock / akshare 数据源(只在已有源基础上加新接口)
- ❌ 不做高频/分钟级数据(保持日级)
- ❌ 不做期权/期货/ETF 申赎数据(后续 v2.x 再考虑)

```python
def fetch_dzjy(date: str | None = None) -> pd.DataFrame | None:
    """大宗交易明细(全市场近 30 日,~2000 条)。
    字段:交易日、证券代码、证券简称、成交价、折溢率、买方营业部、卖方营业部、成交金额。
    用于:识别机构原始股东减持 + 折价率扩大(出货信号)。"""
    import akshare as ak
    if date is None:
        from datetime import datetime
        date = datetime.now().strftime('%Y%m%d')
    return call_with_fallback(
        attempts=[
            (getattr(ak, 'stock_dzjy_mrmxa', None), {'date': date}),
            (getattr(ak, 'stock_dzjy_detail_em', None), {}),
        ],
        cache_key=f'dzjy_{date}',
        timeout=15,
        retries=1,
    )


def fetch_margin(trade_date: str | None = None) -> pd.DataFrame | None:
    """沪深两市融资融券汇总(全市场 ~3000 条/日)。
    字段:代码、简称、融资余额、融券余量、融资买入额、融资偿还额。
    用于:计算个股融资余额 N 日变化率(杠杆资金撤退)。"""
    import akshare as ak
    return call_with_fallback(
        attempts=[
            (getattr(ak, 'stock_margin_underlying_info_szse', None),
             {'date': trade_date} if trade_date else {}),
            (getattr(ak, 'stock_margin_underlying_info_sse', None),
             {'date': trade_date} if trade_date else {}),
        ],
        cache_key=f'margin_{trade_date or "latest"}',
        timeout=15,
        retries=1,
    )


def fetch_holder(date: str | None = None) -> pd.DataFrame | None:
    """股东户数变化(季报披露,全市场 ~5000 条/年)。
    字段:代码、简称、报告期、股东户数、户均持股数、环比变化。
    用于:户数大增 = 筹码分散 = 派发;大降 = 集中 = 吸筹(滞后但准)。"""
    import akshare as ak
    return call_with_fallback(
        attempts=[
            (getattr(ak, 'stock_holder_number_em', None), {}),
        ],
        cache_key='holder_latest',
        timeout=15,
        retries=1,
    )


def fetch_restricted(date: str | None = None) -> pd.DataFrame | None:
    """限售解禁日历(未来 90 日,全市场 ~500 条)。
    字段:代码、简称、解禁日、解禁数量、占总股本比例、流通市值。
    用于:计算 30 日内解禁压力(前瞻卖压)。"""
    import akshare as ak
    return call_with_fallback(
        attempts=[
            (getattr(ak, 'stock_restricted_release_em', None),
             {'date': date} if date else {}),
            (getattr(ak, 'stock_restricted_release_sse_szse', None), {}),
        ],
        cache_key=f'restricted_{date or "forward90"}',
        timeout=15,
        retries=1,
    )
```

```python

- 4 子分独立计分,加权合成 0-100

    'dzjy': 0.30,
    'margin': 0.25,
    'holder': 0.25,
    'restricted': 0.20,
}


def compute_institutional_score(
    code6: str,
    dzjy_df: Optional[pd.DataFrame],
    margin_df: Optional[pd.DataFrame],
    holder_df: Optional[pd.DataFrame],
    restricted_df: Optional[pd.DataFrame],
) -> tuple[int, dict]:
    """
    输入:4 张缓存表
    输出:(综合分 0-100, {子分名: (score, reason)} dict)

    任意子分失败 → 综合分自动重算权重(已实现 _safe_score 装饰器)
    """
    sub_scores = {
        'dzjy': score_dzjy(code6, dzjy_df),
        'margin': score_margin(code6, margin_df),
        'holder': score_holder(code6, holder_df),
        'restricted': score_restricted(code6, restricted_df),
    }
    composite = sum(s * INST_SUB_WEIGHTS[k] for k, (s, _) in sub_scores.items())
    return round(composite), sub_scores
```

**2.3.1 大宗交易子分实现细节**(以 score_dzjy 为例):

```python
def score_dzjy(code6, dzjy_df):
    if dzjy_df is None or dzjy_df.empty:
        return 50, '大宗数据缺失'
    rows = dzjy_df[dzjy_df['_code6'] == code6]
    if rows.empty:
        return 55, '近30日无大宗交易'

    # 折价率列(akshare 列名兼容:折溢价率 / 折溢率)
    rate_col = next((c for c in rows.columns if '折溢' in c), None)
    if not rate_col:
        return 50, '大宗缺折溢率列'
    avg_rate = pd.to_numeric(rows[rate_col], errors='coerce').mean()
    if pd.isna(avg_rate):
        return 50, '大宗折溢率NaN'

    # 基础分(按折价率档位)
    if avg_rate >= 10:     base, desc = 25, f'大宗折价{avg_rate:.1f}%(强出货)'
    elif avg_rate >= 5:    base, desc = 35, f'大宗折价{avg_rate:.1f}%'
    elif avg_rate >= 0:    base, desc = 45, f'大宗折价{avg_rate:.1f}%(微折)'
    elif avg_rate >= -5:   base, desc = 70, f'大宗溢价{abs(avg_rate):.1f}%'
    else:                  base, desc = 80, f'大宗溢价{abs(avg_rate):.1f}%(强接货)'

    # 卖方席位加成:含"机构专用" → -10
    seller_col = next((c for c in rows.columns if '卖方' in c), None)
    if seller_col and rows[seller_col].astype(str).str.contains('机构专用', na=False).any():
        base = max(0, base - 10)
        desc += '+机构席位'

    return base, desc
```
